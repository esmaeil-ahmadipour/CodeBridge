def ask(msg, default=""):
    try:
        v = input(f"{msg}" + (f" [{default}]" if default else "") + ": ").strip()
    except EOFError:
        raise KeyboardInterrupt
    return v or default


def _call(cfg, *argv, pause: bool = True) -> None:
    from ..cli import main
    try:
        main(["--config", str(cfg), *argv])
    except SystemExit:
        pass
    if pause:
        input("\nPress Enter to continue...")


def _edit(spec, cur):
    label, typ, help_, choices = spec
    print(f"\n{label}" + (f"  ({help_})" if help_ else "") + f"\n  current: {cur}")
    if typ == "bool":
        return not cur
    if typ == "choice":
        v = ask(f"  one of {choices}", str(cur))
        return v if v in choices else cur
    if typ == "int":
        v = ask("  number", str(cur))
        return int(v) if v.lstrip("-").isdigit() else cur
    if typ == "list":
        v = ask("  space/comma separated (empty = clear, keep = unchanged)", "keep")
        if v == "keep":
            return cur
        return [x for x in v.replace(",", " ").split() if x]
    return ask("  value", str(cur))


def settings_menu(cfg) -> None:
    while True:
        s = load_settings(cfg, quiet=True)
        rows = [("defaults", k, v) for k, v in SPEC.items()] + [("patch", k, v) for k, v in PATCH_SPEC.items()]
        print("\n== Settings ==")
        for i, (sec, k, sp) in enumerate(rows, 1):
            print(f" {i:>2}. {sp[0]:<34} {s[sec][k]}")
        print("  r. Reset to defaults    0. Back")
        c = ask("Choose")
        if c == "0" or c == "":
            return
        if c == "r":
            _call(cfg, "settings", "reset")
            continue
        if c.isdigit() and 1 <= int(c) <= len(rows):
            sec, k, sp = rows[int(c) - 1]
            s[sec][k] = _edit(sp, s[sec][k])
            save_settings(s, cfg)


def projects_menu(cfg) -> None:
    while True:
        s = load_settings(cfg, quiet=True)
        print("\n== Projects ==")
        for i, p in enumerate(s["projects"], 1):
            print(f" {i}. {p['name']}  {p['path']}")
        print("  a. Add    d. Delete    0. Back")
        c = ask("Choose")
        if c in ("0", ""):
            return
        if c == "a":
            path = ask("Folder path")
            preset = ask(f"Preset {sorted(PRESETS)} (empty = none)")
            argv = ["project", "add", path] + (["--preset", preset] if preset in PRESETS else [])
            _call(cfg, *argv)
        elif c == "d":
            _call(cfg, "project", "remove", ask("Project name"))


MENU = [("1", "Export project for the AI"), ("2", "Apply the AI's diff"), ("3", "Undo last applied diff"),
        ("4", "Projects"), ("5", "Settings"), ("6", "History"), ("0", "Exit")]


def _menu_box(version: str) -> str:
    w = 42
    head = f"─ CodeBridge {version} "
    rows = ["│" + f"  {k}  {t}".ljust(w) + "│" for k, t in MENU]
    return "\n".join(["┌" + head + "─" * (w - len(head)) + "┐", *rows, "└" + "─" * w + "┘"])


def _remember(cfg, target: str) -> None:
    s = load_settings(cfg, quiet=True)
    s["last_project"] = target
    save_settings(s, cfg)


def run_menu(cfg) -> int:
    from .. import get_version
    try:
        while True:
            print("\n" + _menu_box(get_version()))
            try:
                c = input("Type a number, then Enter > ").strip()
            except EOFError:
                return 0
            if c in ("0", "q"):
                return 0
            s = load_settings(cfg, quiet=True)
            if c == "1":
                st = run_screen(s.get("last_project") or ".", s.get("export"), s["projects"])
                if st is None:
                    continue
                target = st["project"]
                isname = any(p["name"] == target for p in s["projects"])
                s["export"] = export_config_to_dict(export_config_from_dict(to_export_dict(st)))
                s["last_project"] = target
                save_settings(s, cfg)
                argv = ["context", "export"] + (["-p", target] if isname else [target]) + (["--changed"] if st["o"] else [])
                _call(cfg, *argv, pause=False)
            elif c in ("2", "3"):
                target = pick_repo(s["projects"], s.get("last_project", ""))
                if target is None:
                    continue
                _remember(cfg, target)
                repo = resolve_repo(s["projects"], target)
                if c == "2":
                    source = ask("\nDiff source: [c]lipboard / [p]aste manually", "c").lower()
                    if source == "c":
                        print("Copy the AI's diff, then press Enter.")
                        input("Press Enter when copied (Ctrl+C to cancel) ")
                        _call(cfg, "patch", "apply", "--clipboard", "-r", repo)
                    elif source == "p":
                        print("\nPaste the complete diff below.")
                        print("Finish by entering CODEBRIDGE_END on a line by itself.")
                        lines = []
                        while True:
                            line = sys.stdin.readline()
                            if not line:
                                break
                            if line.rstrip("\r\n") == "CODEBRIDGE_END":
                                break
                            lines.append(line)

                        diff = "".join(lines)
                        if not diff.strip():
                            print("No diff received. Cancelled.")
                            continue

                        preview = diff.splitlines()
                        print(f"\nReceived {len(preview)} lines.")
                        print("First: " + preview[0][:120])
                        if len(preview) > 1:
                            print("Last : " + preview[-1][:120])
                        print("Full diff is hidden to keep the terminal clean.")

                        with tempfile.NamedTemporaryFile(
                            mode="w", encoding="utf-8", suffix=".diff",
                            prefix="codebridge-", delete=False
                        ) as f:
                            f.write(diff)
                            temp_path = f.name
                        try:
                            _call(cfg, "patch", "apply", temp_path, "-r", repo)
                        finally:
                            from pathlib import Path
                            Path(temp_path).unlink(missing_ok=True)
                    else:
                        print("Invalid choice. Cancelled.")
                else:
                    _call(cfg, "patch", "undo", "-r", repo)
            elif c == "4":
                projects_menu(cfg)
            elif c == "5":
                settings_menu(cfg)
            elif c == "6":
                _call(cfg, "history")
    except KeyboardInterrupt:
        print()
        return 0

# ------------------------------------------------------------------ context
def cmd_context(a, cfg, s) -> int:
    dry = a.sub == "preview"
    targets = []
    if a.project:
        pj = next((x for x in s["projects"] if x["name"] == a.project), None)
        if not pj:
            raise CBError(f"no such project: {a.project}")
        targets = [(pj["name"], pj["path"], project_options(s, pj))]
    elif a.all:
        targets = [(x["name"], x["path"], project_options(s, x)) for x in s["projects"] if x.get("active", True)]
        if not targets:
            raise CBError("no active projects registered (codebridge project add PATH)")
    else:
        root = os.path.abspath(a.target or ".")
        pj = next((x for x in s["projects"] if os.path.abspath(x["path"]) == root), None)
        targets = [(pj["name"] if pj else Path(root).name or "project", root, project_options(s, pj))]
    total = 0
    export_cfg = None
    if s.get("export"):
        try:
            export_cfg = export_config_from_dict(s["export"])
        except Exception:
            export_cfg = None
    for name, root, o in targets:
        if a.compact:
            o["compact"] = True
        # Merge wizard export choices into scanner/exporter options.
        if export_cfg is not None:
            o["include_svg"] = export_cfg.include_svg
            o["exclude_lock_files"] = export_cfg.exclude_lock_files
            o["exclude_test_files"] = export_cfg.exclude_test_files
        only = None
        if a.files:
            only = [f.replace("\\", "/") for f in a.files]
        elif a.changed or a.since:
            if not repo_root(root):
                raise CBError("--changed/--since need a git repository")
            try:
                only = changed_files(root, a.since)
            except ValueError as e:
                raise CBError(str(e))
        r = export_project(name, root, o, outdir=os.getcwd(), output=a.output if len(targets) == 1 else None,
                           dry=dry, backup=not a.no_backup, only=only,
                           clipboard=True if a.clipboard else None,
                           export_cfg=export_cfg)
        if r:
            total += r["files"]
            if not dry:
                history.add(cfg, type="context", project=name, files=r["files"], outs=r["outs"])
    return 0 if total or dry else 1

def _read_source(a):
    src = a.source
    if src == "-":
        return sys.stdin.buffer.read(), "stdin", None
    if src:
        pth = Path(src)
        if not pth.is_file():
            raise CBError(f"diff file not found: {src}")
        return pth.read_bytes(), str(pth), pth
    if not a.clipboard:
        for d in DEFAULT_DIFFS:
            if Path(d).is_file():
                return Path(d).read_bytes(), d, Path(d)
    clip = from_clipboard()
    if clip and clip.strip():
        return clip, "clipboard", None
    raise CBError("no diff given: pass a file, pipe to stdin ('-'), or copy the AI reply first")


def _show_plan(plan, preview=False) -> None:
    print(f"Repo  : {plan.root}")
    print(f"Files : {len(plan.patches)}")
    for fp in plan.patches:
        st = plan.per_file.get(fp.name, ("ok", ""))[0]
        tag = {"ok": "ok", "already": "already applied", "fail": "CANNOT APPLY"}[st]
        print(f"  [{fp.kind}] {fp.name}  (+{fp.added} -{fp.removed})  {tag}")
        if st == "fail" and plan.per_file[fp.name][1]:
            print("        " + plan.per_file[fp.name][1].replace("\n", "\n        "))
    if plan.status == "apply" and plan.flags:
        print(f"Note  : applies in tolerant mode ({' '.join(plan.flags)})")
    if plan.status == "fail" and plan.error and not any(s == "fail" for s, _ in plan.per_file.values()):
        print(f"Error : {plan.error}")


def cmd_patch(a, cfg, s) -> int:
    ps = s["patch"]
    if a.sub == "undo":
        root = require_repo(a.repo or ".")
        meta = engine.undo(root, a.run, a.force)
        print(f"Reverted run {meta['id']} ({len(meta['touched'])} file(s)) in {root}")
        history.add(cfg, type="undo", repo=root, run=meta["id"])
        return 0
    root = require_repo(a.repo or ".")
    raw, label, path = _read_source(a)
    text = clean_diff(raw)
    plan = engine.analyze(root, text)
    print(f"Source: {label}")
    _show_plan(plan)
    if a.sub == "preview" and a.diff:
        print("\n" + plan.text)
    if plan.status == "already":
        print("\nAlready applied: nothing to do.")
        return 0
    if a.sub in ("check", "preview"):
        print("\nOK: the diff can be applied." if plan.status == "apply" else "\nThe diff cannot be fully applied.")
        return 0 if plan.status == "apply" else 2
    # ---- apply
    partial = a.partial or not ps["strict"]
    dirty = git_status(root)
    if dirty:
        print("\nwarning: the repo has uncommitted changes (they stay untouched).")
    if plan.status == "fail" and not partial:
        _fail(plan, cfg, s, a, "strict mode: nothing was changed")
        return 2
    if plan.flags:
        print("\nWARNING: Git could only accept this diff using tolerant options.")
        print("The patch may have incomplete or unusual hunk context.")
        print("Press Enter to apply, or type c to cancel.")
        if not sys.stdin.isatty():
            raise CBError("tolerant patch requires interactive confirmation")
        answer = input("Apply tolerant patch? [Enter/c] ").strip().lower()
        if answer == "c":
            print("Cancelled. Nothing was changed.")
            return 0
        if answer != "":
            print("Invalid choice. Cancelled. Nothing was changed.")
            return 0
    if ps["confirm_before_apply"] and not a.yes:
        if not sys.stdin.isatty():
            raise CBError("no terminal to confirm; re-run with -y")
        if input("\nApply these changes? [Y/n] ").strip().lower() not in ("", "y", "yes"):
            print("Cancelled. Nothing was changed.")
            return 0
    rid_hint = time.strftime("%Y%m%d-%H%M%S")
    branch = a.branch or (f"codebridge/{rid_hint}" if ps["create_branch"] else None)
    commit = a.commit or (f"Apply AI patch (codebridge {rid_hint})" if ps["auto_commit"] else None)
    try:
        meta = engine.apply_plan(plan, partial=partial, branch=branch, commit=commit, source=label)
    except CBError as e:
        if e.code == 2:
            _fail(plan, cfg, s, a, None)
        raise
    print(f"\nDone: {len(meta['files'])} applied, {len(meta['skipped'])} skipped. Run {meta['id']}")
    for n in meta["notes"]:
        print("warning:", n)
    if meta["branch"]:
        print(f"Branch: {meta['branch']}")
    if meta["commit"]:
        print(f"Committed: {meta['commit']}")
    print(f"Undo with: codebridge patch undo -r \"{root}\"")
    history.add(cfg, type="patch", repo=root, run=meta["id"], files=[f["name"] for f in meta["files"]])
    if path and ps["archive_applied"] and not a.no_archive and not meta["skipped"]:
        dst = path.with_name(f"{path.name}.applied-{time.strftime('%Y%m%d-%H%M%S')}")
        shutil.move(str(path), str(dst))
        print(f"Archived: {dst.name}")
    return 0


def git_status(root) -> str:
    from .core.git import git
    return git(root, "status", "--porcelain").stdout.strip()


def _fail(plan, cfg, s, a, headline) -> None:
    if headline:
        print(f"\n{headline}")
    if getattr(a, "no_fix", False):
        return
    prompt = build_fix_prompt(plan, s["defaults"])
    out = Path.cwd() / "codebridge-fix.txt"
    out.write_text(prompt, encoding="utf-8")
    clip = to_clipboard(prompt)
    print(f"Fix prompt for the AI written to {out.name}" + (" and copied to the clipboard." if clip else "."))
    print("Send it to the AI, then apply the new diff.")
    history.add(cfg, type="patch-failed", repo=plan.root, files=[fp.name for fp in plan.patches])

def cmd_project(a, cfg, s) -> int:
    if a.sub == "list" or not a.sub:
        if not s["projects"]:
            print("No projects registered. Add one: codebridge project add PATH")
        for x in s["projects"]:
            print(f"  {x['name']}  {x['path']}" + ("" if x.get("active", True) else "  (inactive)"))
    elif a.sub == "add":
        path = os.path.abspath(a.path)
        if not os.path.isdir(path):
            raise CBError(f"folder not found: {path}")
        name = a.name or Path(path).name
        if any(x["name"] == name for x in s["projects"]):
            raise CBError(f"a project named '{name}' already exists")
        s["projects"].append({"name": name, "path": path, "active": True,
                              "overrides": dict(PRESETS[a.preset]) if a.preset else {}})
        save_settings(s, cfg)
        print(f"Added {name}")
    elif a.sub == "remove":
        n = len(s["projects"])
        s["projects"] = [x for x in s["projects"] if x["name"] != a.name]
        if len(s["projects"]) == n:
            raise CBError(f"no such project: {a.name}")
        save_settings(s, cfg)
        print(f"Removed {a.name}")
    return 0


def cmd_history(a, cfg, s) -> int:
    rows = history.read(cfg, a.n)
    if not rows:
        print("No history yet.")
    for r in rows:
        t = r.get("type", "?")
        detail = r.get("project") or r.get("repo") or ""
        extra = f" run={r['run']}" if r.get("run") else ""
        n = r.get("files")
        n = f" files={len(n) if isinstance(n, list) else n}" if n is not None else ""
        print(f"{r.get('time', '')}  {t:<13} {detail}{extra}{n}")
    return 0


def cmd_settings(a, cfg, s) -> int:
    import json
    if a.sub == "path":
        print(cfg)
    elif a.sub == "reset":
        from .core.config import default_settings
        if Path(cfg).is_file():
            shutil.copy(cfg, str(cfg) + ".bak")
        d = default_settings()
        d["projects"] = s["projects"]
        save_settings(d, cfg)
        print(f"Settings reset to defaults (old file saved as {Path(cfg).name}.bak, projects kept)")
    else:
        print(json.dumps(s, indent=2, ensure_ascii=False))
    return 0


def main(argv=None) -> int:
    setup_console()
    ap = build_parser()
    a = ap.parse_args(argv)
    try:
        cfg = resolve_config(a.config)
        if not a.cmd:
            if sys.stdin.isatty() and sys.stdout.isatty():
                from .ui.menu import run_menu
                return run_menu(cfg)
            ap.print_help()
            return 0
        s = load_settings(cfg)
        if a.cmd in ("context", "patch", "project", "settings") and not getattr(a, "sub", None):
            if a.cmd in ("project", "settings"):
                a.sub = "list" if a.cmd == "project" else "show"
            else:
                ap.parse_args([a.cmd, "-h"])
        fn = {"context": cmd_context, "patch": cmd_patch, "project": cmd_project,
              "history": cmd_history, "settings": cmd_settings}[a.cmd]
        return fn(a, cfg, s)
    except CBError as e:
        print(f"error: {e}", file=sys.stderr)
        return e.code
    except KeyboardInterrupt:
        print("\ncancelled", file=sys.stderr)
        return 130

def migrate(raw: dict) -> dict:
    """Accept CodeDump v1/v2 or CodeBridge v3 settings; return v3 filled with defaults."""
    s = default_settings()
    d = _norm_opts(raw.get("defaults", raw.get("global")))
    # Mandatory security: never allow legacy settings to disable redaction.
    d.pop("redact_secrets", None)
    bk = d.pop("backup", {})
    s["defaults"].update(d)
    s["defaults"]["backup"].update(bk)
    s["defaults"]["redact_secrets"] = True
    # Preserve export config block if present; otherwise absent (None = unanswered).
    if "export" in raw and isinstance(raw["export"], dict):
        s["export"] = raw["export"]
    if isinstance(raw.get("last_project"), str):
        s["last_project"] = raw["last_project"]
    s["patch"].update({k: v for k, v in (raw.get("patch") or {}).items() if k in PATCH_DEFAULTS})
    for p in raw.get("projects") or []:
        ov = {k: v for k, v in p.items() if k in OLD_KEYS or k in DEFAULTS}
        ov.pop("redact_secrets", None)
        ov.update(p.get("overrides") or p.get("options") or {})
        ov.pop("redact_secrets", None)
        s["projects"].append({"name": p.get("name") or Path(p.get("path", "x")).name,
                              "path": p.get("path", ""),
                              "active": p.get("active", p.get("enabled", True)),
                              "overrides": _norm_opts(ov)})

    return s

"""Analyze, apply (atomic) and undo AI diffs. Each run is recorded under .git/codebridge/runs/<id>/."""
from __future__ import annotations
import contextlib, hashlib, json, os, shutil, tempfile, time
from dataclasses import dataclass, field
from pathlib import Path
from ..core.git import git, git_dir
from ..core.util import CBError, run

# Tried in order; the first that works is used (and reported).
MODES = [(), ("--recount",), ("--recount", "--ignore-whitespace"),
         ("--recount", "--ignore-whitespace", "-C1"),
         ("--recount", "--ignore-whitespace", "-C0", "--unidiff-zero")]


@dataclass
class Plan:
    root: str
    text: str
    p: int
    patches: list
    status: str                      # apply | already | fail
    flags: tuple = ()
    per_file: dict = field(default_factory=dict)   # name -> (ok|already|fail, error)
    error: str = ""


@contextlib.contextmanager
def _patch_file(text):
    d = tempfile.mkdtemp(prefix="cb-")
    try:
        p = os.path.join(d, "p.diff")
        Path(p).write_bytes(text.encode("utf-8"))
        yield p
    finally:
        shutil.rmtree(d, ignore_errors=True)


def _apply(root, path, p, flags=(), check=False, reverse=False):
    cmd = ["git", "-C", str(root), "apply", f"-p{p}", "--whitespace=nowarn", *flags]
    if check:
        cmd.append("--check")
    if reverse:
        cmd.append("-R")
    cmd.append(path)
    r = run(cmd)
    return r.returncode == 0, (r.stderr or "").strip()


def _find_mode(root, path, p, reverse=False):
    err = ""
    for flags in MODES:
        ok, err = _apply(root, path, p, flags, check=True, reverse=reverse)
        if ok:
            return flags, ""
    return None, err


def analyze(root, text) -> Plan:
    patches = split_patches(text)
    if not patches:
        raise CBError("no per-file changes found in the diff")
    p = detect_p(text)
    with _patch_file(text) as f:
        flags, err = _find_mode(root, f, p)
        if flags is not None:
            return Plan(root, text, p, patches, "apply", flags,
                        {fp.name: ("ok", "") for fp in patches})
        rflags, _ = _find_mode(root, f, p, reverse=True)
        if rflags is not None:
            return Plan(root, text, p, patches, "already", rflags,
                        {fp.name: ("already", "") for fp in patches})
    per = {}
    for fp in patches:
        with _patch_file(fp.text) as f1:
            fl, e = _find_mode(root, f1, p)
            if fl is not None:
                per[fp.name] = ("ok", "")
            else:
                rf, _ = _find_mode(root, f1, p, reverse=True)
                per[fp.name] = ("already", "") if rf is not None else ("fail", e)
    return Plan(root, text, p, patches, "fail", (), per, err)


# ---------------------------------------------------------------- runs
def _hash(path: Path):
    return hashlib.sha256(path.read_bytes()).hexdigest() if path.is_file() else None


def runs_dir(root) -> Path:
    return Path(git_dir(root)) / "codebridge" / "runs"


def list_runs(root) -> list:
    d = runs_dir(root)
    out = []
    if d.is_dir():
        for r in sorted(d.iterdir()):
            m = r / "meta.json"
            if m.is_file():
                out.append(json.loads(m.read_text(encoding="utf-8")))
    return out


def _new_id(base: Path) -> str:
    rid = time.strftime("%Y%m%d-%H%M%S")
    n, cand = 1, rid
    while (base / cand).exists():
        n += 1
        cand = f"{rid}-{n}"
    return cand


def _touched(patches) -> list:
    names = set()
    for fp in patches:
        for n in (fp.old, fp.new):
            if n:
                names.add(n)
    return sorted(names)


def preflight(root, patches, branch=None, commit=None) -> None:
    """Fail early (before touching files) on problems that would otherwise appear after."""
    if branch:
        if git(root, "check-ref-format", "--branch", branch).returncode != 0:
            raise CBError(f"invalid branch name: {branch}")
        if git(root, "rev-parse", "--verify", "-q", f"refs/heads/{branch}").returncode == 0:
            raise CBError(f"branch already exists: {branch}")
    if commit:
        dirty = git(root, "status", "--porcelain", "--", *_touched(patches)).stdout.strip()
        if dirty:
            raise CBError("these files already have uncommitted changes, so --commit would include "
                          "them too; commit/stash them first or run without --commit:\n"
                          + "\n".join("  " + l for l in dirty.splitlines()))


def apply_plan(plan: Plan, partial=False, branch=None, commit=None, source="") -> dict:
    root, patches = plan.root, plan.patches
    if plan.status == "fail":
        if not partial:
            raise CBError("nothing was changed: some files cannot be applied (strict mode)", 2)
        patches = [fp for fp in patches if plan.per_file.get(fp.name, ("fail",))[0] == "ok"]
        if not patches:
            raise CBError("nothing can be applied", 2)
    text = plan.text if patches is plan.patches else "".join(fp.text for fp in patches)
    preflight(root, patches, branch, commit)
    touched = _touched(patches)
    base = runs_dir(root)
    rid = _new_id(base)
    rdir = base / rid
    before = {}
    for n in touched:
        f = Path(root) / n
        before[n] = _hash(f)
        if f.is_file():
            dst = rdir / "before" / n
            dst.parent.mkdir(parents=True, exist_ok=True)
            shutil.copy2(f, dst)
    rdir.mkdir(parents=True, exist_ok=True)
    with _patch_file(text) as pf:
        flags, err = _find_mode(root, pf, plan.p)
        if flags is None:
            shutil.rmtree(rdir, ignore_errors=True)
            raise CBError(f"nothing was changed: {err}", 2)
        ok, err = _apply(root, pf, plan.p, flags)
        if not ok:
            shutil.rmtree(rdir, ignore_errors=True)
            raise CBError(f"nothing was changed: {err}", 2)
    after = {n: _hash(Path(root) / n) for n in touched}
    meta = {"id": rid, "time": time.strftime("%Y-%m-%d %H:%M:%S"), "source": source,
            "flags": list(flags), "p": plan.p, "touched": touched, "before": before,
            "after": after, "files": [{"name": fp.name, "kind": fp.kind} for fp in patches],
            "skipped": [n for n, (s, _) in plan.per_file.items() if s != "ok"] if partial else [],
            "undone": None, "branch": None, "commit": None}
    (rdir / "patch.diff").write_text(text, encoding="utf-8")
    notes = []
    if branch:
        r = git(root, "checkout", "-b", branch)
        if r.returncode == 0:
            meta["branch"] = branch
        else:
            notes.append(f"could not create branch: {r.stderr.strip()}")
    if commit:
        git(root, "add", "-A", "--", *touched)
        r = git(root, "commit", "-q", "-m", commit)
        if r.returncode == 0:
            meta["commit"] = commit
        else:
            notes.append(f"commit failed: {(r.stderr or r.stdout).strip()}")
    (rdir / "meta.json").write_text(json.dumps(meta, indent=2), encoding="utf-8")
    meta["notes"] = notes
    return meta


def undo(root, run_id=None, force=False) -> dict:
    runs = [r for r in list_runs(root) if not r["undone"]]
    if run_id:
        runs = [r for r in list_runs(root) if r["id"] == run_id]
        if not runs:
            raise CBError(f"no such run: {run_id}", 3)
        if runs[0]["undone"]:
            raise CBError(f"run {run_id} was already undone", 3)
    if not runs:
        raise CBError("nothing to undo (no recorded runs)", 3)
    meta = runs[-1]
    rdir = runs_dir(root) / meta["id"]
    changed = [n for n in meta["touched"] if _hash(Path(root) / n) != meta["after"][n]]
    if changed and not force:
        raise CBError("files changed since this run, nothing was reverted "
                      "(use --force to restore the pre-run versions anyway):\n"
                      + "\n".join("  " + n for n in changed), 3)
    for n in meta["touched"]:
        f = Path(root) / n
        snap = rdir / "before" / n
        if meta["before"][n] is None:
            if f.is_file():
                f.unlink()
            p = f.parent
            while p != Path(root) and p.is_dir() and not any(p.iterdir()):
                p.rmdir()
                p = p.parent
        else:
            f.parent.mkdir(parents=True, exist_ok=True)
            shutil.copy2(snap, f)
    meta["undone"] = time.strftime("%Y-%m-%d %H:%M:%S")
    (rdir / "meta.json").write_text(json.dumps(meta, indent=2), encoding="utf-8")
    return meta

def sh(cwd, *cmd):
    r = subprocess.run(cmd, cwd=cwd, capture_output=True, text=True, env=ENV, encoding="utf-8")
    assert r.returncode == 0, (cmd, r.stderr)
    return r.stdout


def make_repo(files: dict) -> Path:
    d = Path(tempfile.mkdtemp())
    sh(d, "git", "init", "-q")
    sh(d, "git", "config", "core.autocrlf", "false")
    for n, c in files.items():
        f = d / n
        f.parent.mkdir(parents=True, exist_ok=True)
        with f.open("w", encoding="utf-8", newline="") as stream:
            stream.write(c)
    sh(d, "git", "add", "-A")
    sh(d, "git", "commit", "-q", "-m", "init")
    return d


def make_diff(repo: Path, edits: dict) -> str:
    """Apply edits (None = delete), capture `git diff`, then restore the tree."""
    for n, c in edits.items():
        f = repo / n
        if c is None:
            f.unlink()
        else:
            f.parent.mkdir(parents=True, exist_ok=True)
            with f.open("w", encoding="utf-8", newline="") as stream:
                stream.write(c)
    sh(repo, "git", "add", "-A")
    d = sh(repo, "git", "diff", "--cached", "--no-color")
    sh(repo, "git", "reset", "-q", "--hard")
    sh(repo, "git", "clean", "-fdq")
    return d

import contextlib, io, os, re, tempfile, unittest
from pathlib import Path
from unittest import mock
import helpers
from helpers import make_repo, make_diff
from codebridge.cli import main


def call(*argv):
    out, err = io.StringIO(), io.StringIO()
    with contextlib.redirect_stdout(out), contextlib.redirect_stderr(err):
        try:
            code = main(list(argv))
        except SystemExit as e:
            code = e.code
    return code, out.getvalue(), err.getvalue()


class TTY(io.StringIO):
    def isatty(self):
        return True


class T(unittest.TestCase):
    def setUp(self):
        self.cfg = str(Path(tempfile.mkdtemp()) / "settings.json")
        self.r = make_repo({"a.txt": "one\ntwo\nthree\nfour\nfive\nsix\n", "m.py": "x=1\n"})
        self.cwd = os.getcwd()
        self.work = tempfile.mkdtemp()
        os.chdir(self.work)

    def tearDown(self):
        os.chdir(self.cwd)

    def test_version(self):
        self.assertEqual(call("--version")[0], 0)

    def test_context_export_changed(self):
        (self.r / "m.py").write_text("x=2\n")
        c, out, _ = call("--config", self.cfg, "context", "export", str(self.r), "--changed", "--no-backup")
        self.assertEqual(c, 0)
        rep = Path(self.work, f"code_report_{self.r.name}.txt").read_text(encoding="utf-8")
        self.assertIn("m.py", rep); self.assertNotIn("FILE:  a.txt", rep)

    def test_patch_full_cycle(self):
        diff = make_diff(self.r, {"a.txt": "one\ntwo\n3\nfour\nfive\nsix\n"})
        d = Path(self.work, "diff.patch"); d.write_text(diff, encoding="utf-8")
        c, out, err = call("--config", self.cfg, "patch", "check", str(d), "-r", str(self.r))
        self.assertEqual(c, 0, out + err)
        c, out, err = call("--config", self.cfg, "patch", "apply", str(d), "-r", str(self.r), "-y")
        self.assertEqual(c, 0, out + err)
        self.assertIn("3", (self.r / "a.txt").read_text())
        self.assertFalse(d.exists()); self.assertTrue(list(Path(self.work).glob("diff.patch.applied-*")))
        c, out, _ = call("--config", self.cfg, "history")
        self.assertIn("patch", out)
        c, out, err = call("--config", self.cfg, "patch", "undo", "-r", str(self.r))
        self.assertEqual(c, 0, err); self.assertIn("two\nthree", (self.r / "a.txt").read_text())

    def test_failure_writes_fix_prompt(self):
        diff = make_diff(self.r, {"a.txt": "one\ntwo\n3\nfour\nfive\nsix\n"})
        (self.r / "a.txt").write_text("different\n")
        d = Path(self.work, "p.diff"); d.write_text(diff, encoding="utf-8")
        c, out, err = call("--config", self.cfg, "patch", "apply", str(d), "-r", str(self.r), "-y")
        self.assertEqual(c, 2)
        self.assertIn("FIX REQUEST", Path(self.work, "codebridge-fix.txt").read_text(encoding="utf-8"))
        self.assertEqual((self.r / "a.txt").read_text(), "different\n")

    def test_not_a_repo_and_missing_diff(self):
        c, _, err = call("--config", self.cfg, "patch", "check", "nope.diff", "-r", self.work)
        self.assertEqual(c, 1)

    def test_project_registry(self):
        self.assertEqual(call("--config", self.cfg, "project", "add", str(self.r), "-n", "demo", "--preset", "python")[0], 0)
        c, out, _ = call("--config", self.cfg, "project", "list")
        self.assertIn("demo", out)
        c, out, _ = call("--config", self.cfg, "context", "preview", "-p", "demo")
        self.assertEqual(c, 0); self.assertIn("m.py", out)
        self.assertEqual(call("--config", self.cfg, "project", "remove", "demo")[0], 0)

    def test_tolerant_apply_requires_confirmation_even_with_yes(self):
        original = "one\ntwo\nthree\nfour\nfive\nsix\n"
        diff = make_diff(self.r, {"a.txt": "one\ntwo\n3\nfour\nfive\nsix\n"})
        hunk = next(line for line in diff.splitlines() if line.startswith("@@ "))
        bad_hunk = re.sub(r"(@@ -\d+),\d+( \+\d+),\d+( @@)", r"\1,1\2,1\3", hunk)
        bad = diff.replace(hunk, bad_hunk)
        patch_file = Path(self.work, "tolerant.diff")
        patch_file.write_text(bad, encoding="utf-8")

        with mock.patch("codebridge.cli.sys.stdin", TTY("c\n")):
            c, out, err = call("--config", self.cfg, "patch", "apply",
                               str(patch_file), "-r", str(self.r), "-y")

        self.assertEqual(c, 0, out + err)
        self.assertIn("tolerant", out.lower())
        self.assertIn("Cancelled", out)
        self.assertEqual((self.r / "a.txt").read_text(), original)
